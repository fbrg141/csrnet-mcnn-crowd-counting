\begin{frame}{Problem i istraživačko pitanje}
\begin{itemize}
\item Okluzija i male glave otežavaju detekciju pojedinačnih osoba u gustim scenama.
\item Regresija mape gustine: broj ljudi je suma predviđene mape.
\item Poredimo MCNN i CSRNet na ShanghaiTech Part A u budžetu od 50 epoha.
\item Pitanje: koja \emph{konfiguracija} daje manju grešku, i koliko MCNN zavisi od podešavanja i seed-a?
\end{itemize}
\begin{block}{Obuhvat dokaza}Završena MCNN pretraga i potvrde; završen CSRNet samo za seed 42. Nema rezultata za Part B.\end{block}
\end{frame}

\begin{frame}{Podaci i zajednički protokol}
\begin{itemize}
\item Part A: 300 zvaničnih trening slika i 182 test slike.
\item Trening/validacija: 270/30, poslednjih 10\% \textbf{leksikografski sortiranih} naziva; nije nasumičan uzorak.
\item Ulaz $768\times1024$; 50 epoha, batch 4, SGD, srednji pikselni MSE; trening odbacuje poslednji nepotpun batch.
\item Bez augmentacije, rasporeda stope učenja i ranog zaustavljanja.
\item Checkpoint: najmanji validacioni MAE, ne poslednja epoha po automatizmu.
\end{itemize}
\small MAE $=\frac1n\sum_i|e_i|$; RMSE $=\sqrt{\frac1n\sum_i e_i^2}$.\\
RMSE jače kažnjava velike greške; nije varijabilnost između seed-ova.
\end{frame}

\begin{frame}{MCNN: validaciona pretraga i potvrda}
\begin{itemize}
\item Mreža: lr $\{10^{-6},3\cdot10^{-6},10^{-5}\}$, momentum $\{0.90,0.95\}$, weight decay $\{0,10^{-4}\}$.
\item Svih 12 konfiguracija: seed 42, po 50 epoha.
\item Početna: $(10^{-6},0.95,0)$; izabrana: $(10^{-5},0.95,10^{-4})$.
\item Izbor samo po validacionom MAE; četiri dodatna treninga potvrđuju obe konfiguracije na seed-ovima 123 i 2026.
\item CSRNet nije podešavan: $(10^{-5},0.95,0)$, samo seed 42.
\end{itemize}
\begin{alertblock}{Granica interpretacije}Test nije korišćen za rangiranje ove pretrage, ali pilot test rezultati su ranije viđeni. Test nije globalno netaknut.\end{alertblock}
\end{frame}

\item Početni MCNN je nedovoljno istreniran u ovom budžetu.
\item Jedan CSRNet seed ne dokazuje opštu superiornost ili veću stabilnost.
\end{itemize}
\end{frame}

\begin{frame}{Poreklo dokaza i reprodukcija}
\begin{itemize}
\item MCNN arhiva: 12 pokušaja + 4 potvrde, 6 test zapisa; provereno 48 hash-eva izveštaja. Nema MCNN težina u arhivi.
\item CSRNet lokalno: 50 epoha, RTX 4070, CUDA FP32; trening 1583.2287 s. Nije merenje inferencije.
\item Originalni laki JSON/CSV: \texttt{reports/evidence/}; dataset i težine nisu uključeni.
\item \texttt{scripts/plot\_training\_curves.py}: provera hash-eva, izvođenje tabela, SD i sva četiri grafikona.
\item Izveštaj, README i beleške navode isti završeni obuhvat.
\end{itemize}
\end{frame}

\begin{frame}{Ograničenja i sledeći eksperimenti}
\begin{itemize}
\item Mali, nerandomizovan validacioni skup; ranije viđeni pilot test rezultati.
\item Samo Part A; CSRNet samo seed 42; nema testa značajnosti.
\item MCNN dodatno podešavan; jednak broj epoha nije jednak ukupan trošak pretrage.
\item Razlike od literature: fiksne mape, bez augmentacije, ograničen trening. Uticaji nisu izolovani ablacijama i ne moraju biti jednaki za modele.
\item Sledeće: dodatni CSRNet seed-ovi, Part B, duži trening, augmentacija, adaptivne mape, raspored stope učenja i posebno merenje inferencije.
\end{itemize}
\end{frame}

\begin{frame}{Zaključak}
\begin{block}{Šta podaci podržavaju?}CSRNet ima niže test greške od podešenog MCNN-a za seed 42 u izvedenom Part A protokolu.\end{block}
\begin{itemize}
\item MCNN podešavanje mnogo menja rezultat u budžetu od 50 epoha.
\item Troseed MCNN analiza prikazuje varijabilnost, a ne samo najbolji rezultat.
\item Prednost weight decay od 0.0135 MAE je zanemarljiva kao dokaz koristi.
\item Nema zaključka o opštoj superiornosti, značajnosti ili stabilnosti CSRNet-a bez dodatnih eksperimenata.
\end{itemize}
